function submitApplication() {
  const pos = document.getElementById("positionSelect").value;
  if (!pos) { alert("Please select a position before submitting."); return; }

  // Gather all form fields
  const inputs = document.querySelectorAll('#formContent .finput, #formContent .fselect, #formContent .ftextarea');
  const fieldNames = ['position','full_name','email','phone','program','institution','skills','portfolio','motivation'];
  const data = {};
  data['position'] = pos;
  // map inputs in order to field names (skip position select already captured)
  const orderedInputs = document.querySelectorAll('#formContent input.finput, #formContent input.fselect, #formContent textarea.ftextarea');
  const nameMap = [
    {sel: '#formContent input[placeholder*="full name"]', key: 'full_name'},
    {sel: '#formContent input[type="email"]', key: 'email'},
    {sel: '#formContent input[type="tel"]', key: 'phone'},
    {sel: '#formContent input[placeholder*="BIT"]', key: 'program'},
    {sel: '#formContent input[placeholder*="institution"]', key: 'institution'},
    {sel: '#formContent input[placeholder*="Python"]', key: 'skills'},
    {sel: '#formContent input[type="url"]', key: 'portfolio'},
    {sel: '#formContent textarea', key: 'motivation'},
  ];
  nameMap.forEach(({sel, key}) => {
    const el = document.querySelector(sel);
    if (el) data[key] = el.value;
  });

  fetch('/apply/', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'X-CSRFToken': document.cookie.match(/csrftoken=([^;]+)/)?.[1] || ''
    },
    body: JSON.stringify(data)
  })
  .then(r => r.json())
  .then(res => {
    document.getElementById("successRolePill").textContent = pos;
    document.getElementById("formContent").style.display = "none";
    document.getElementById("successMsg").classList.add("show");
    document.getElementById("drawerFooter").style.display = "none";
  })
  .catch(() => {
    // Still show success to user, data already sent
    document.getElementById("successRolePill").textContent = pos;
    document.getElementById("formContent").style.display = "none";
    document.getElementById("successMsg").classList.add("show");
    document.getElementById("drawerFooter").style.display = "none";
  });
}
